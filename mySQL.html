4. MySQL - Todo lo del pizarrón

A) Crear BD

CREATE DATABASE escuela;
USE escuela;

B) Crear Tablas

CREATE TABLE alumnos (
  id INT PRIMARY KEY AUTO_INCREMENT,
  nombre VARCHAR(50),
  carrera VARCHAR(50),
  edad INT
);

C) Registros (INSERT)

INSERT INTO alumnos (nombre, carrera, edad) VALUES
('Juan Perez', 'Informatica', 18),
('Maria Lopez', 'Sistemas', 19),
('Carlos Ruiz', 'Informatica', 20);

D) Consultas

-- Ver todos
SELECT * FROM alumnos;

-- Filtrar
SELECT * FROM alumnos WHERE carrera = 'Informatica';

-- Ordenar
SELECT * FROM alumnos ORDER BY edad DESC;

-- Buscar
SELECT nombre FROM alumnos WHERE edad > 18;
